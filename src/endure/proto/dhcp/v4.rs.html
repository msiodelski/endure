1//! `dhcp::v4` is a module providing DHCPv4 message parsing capabilities.
2//!
3//! The DHCPv4 protocol described in <https://www.ietf.org/rfc/rfc2131.html>
4//! has been built on top of the BOOTP protocol. It reuses its message structures
5//! extending them with the DHCP options carrying additional configuration data.
6//! This module internally calls the [crate::proto::bootp] module to parse the
7//! fixed fields of the DHCPv4 messages.
8//!
9//! This module provides functions to parse DHCPv4 options carried in the messages.

294/// Raw packet state.
295///
296/// It is a default state of the received DHCPv4 packets. A packet in this
297/// state includes an unparsed buffer. The packet must be converted into the
298/// parsable state with the [ReceivedPacket::into_parsable] to parse the
299/// packet.
300pub struct RawState {
301    /// Unparsed packet data.
302    data: Vec<u8>,
303}
304
305/// Partially parsed packet state.
306///
307/// A received packet must be transitioned to this state with the
308/// [ReceivedPacket::into_parsable] before it can be parsed.
309///
310/// ### Selective Parsing
311///
312/// It is possible to access individual values in the received DHCP message
313/// without parsing the entire packet. The respective functions of the [ReceivedPacket]
314/// access the DHCP fields at their offsets and cache the read values in the
315/// [PartiallyParsedState]. The cached value is returned the next time the same
316/// function is called. Selective parsing improves performance when the caller
317/// is only interested in accessing the portions of a packet.
318#[derive(Clone, Debug)]
319pub struct PartiallyParsedState {
320    bootp: bootp::ReceivedPacket<bootp::PartiallyParsedState>,
321    options: HashMap<u8, ReceivedOption>,
322    options_cursor: u32,
323}
324
325/// A structure representing a received DHCP packet.
326///
327/// # State Model
328///
329/// A received packet's state is represented by a generic data type `State`.
330/// It allows for exposing different set of functions depending on the packet
331/// state.
332///
333/// A received packet is initially in the [RawState] state. In this state, the packet
334/// is unparsed and exposes no data parsing functions. The packet must be
335/// explicitly transitioned to the [`PartiallyParsedState`] before parsing and
336/// accessing the named data fields carried in the packet.
337#[derive(Clone, Debug)]
338pub struct ReceivedPacket<State> {
339    /// Packet state.
340    state: State,
341}
342
343/// A shorthand type for the raw packet.
344pub type RawPacket = ReceivedPacket<RawState>;
345
346/// A shorthand type for the partially parsed packet.
347pub type PartiallyParsedPacket = ReceivedPacket<PartiallyParsedState>;
348
349/// A shareable and lockable instance of the [`PartiallyParsedPacket`].
350pub type SharedPartiallyParsedPacket = Arc<RwLock<PartiallyParsedPacket>>;
351
352impl ReceivedPacket<RawState> {
353    /// Creates a new raw packet instance.
354    ///
355    /// # Parameters
356    ///
357    /// - `data` is a reference to the buffer holding the packet.
358    pub fn new(data: &[u8]) -> ReceivedPacket<RawState> {
359        ReceivedPacket {
360            state: RawState {
361                data: data.to_vec(),
362            },
363        }
364    }
365
366    /// Converts the packet to the BOOTP packet.
367    pub fn as_bootp(self) -> bootp::ReceivedPacket<bootp::RawState> {
368        bootp::ReceivedPacket::new(&self.state.data)
369    }
370
371    /// Transitions the packet from the [`RawState`] to the [`PartiallyParsedState`].
372    ///
373    /// # Result
374    ///
375    /// It returns non-shared, non-lockable, parsable packet instance.
376    pub fn into_parsable(&self) -> ReceivedPacket<PartiallyParsedState> {
377        let bootp = bootp::ReceivedPacket::new(&self.state.data);
378        ReceivedPacket::<PartiallyParsedState> {
379            state: PartiallyParsedState {
380                bootp: bootp.into_parsable(),
381                options: HashMap::new(),
382                options_cursor: OPTIONS_POS,
383            },
384        }
385    }
386
387    /// Transitions the packet from the [`RawState`] to the [`PartiallyParsedState`].
388    ///
389    /// # Result
390    ///
391    /// It returns shared, lockable, parsable packet instance.
392    pub fn into_shared_parsable(&self) -> SharedPartiallyParsedPacket {
393        Arc::new(RwLock::new(self.into_parsable()))
394    }
395}

480    /// Attempts to read DHCP option from buffer at specified position.
481    ///
482    /// If the read is successful the `pos` value is increased by the
483    /// amount of data read. Otherwise, the `pos` value remains unchanged.
484    /// This function does not validate the length or format of the option.
485    /// It may, however, return `BufferError` when the decoded option
486    /// length goes beyond the size of the buffer.
487    ///
488    /// # Parameters
489    ///
490    /// - `buf` is a buffer to read from.
491    /// - `pos` is a position in the buffer to start reading.
492    fn try_read_option(
493        buf: &mut ReceiveBuffer,
494        pos: &mut u32,
495    ) -> Result<ReceivedOption, BufferError> {
496        let parsed_code = buf.read_u8(*pos)?;
497        match buf.read_u8(*pos)? {
498            // These options have a special format. They only contain the
499            // option code and no option length.
500            OPTION_CODE_PAD | OPTION_CODE_END => {
501                *pos += 1;
502                Ok(ReceivedOption::new(parsed_code, &Vec::new()))
503            }
504            _ => match buf.read_u8(*pos + 1)? {
505                0 => {
506                    *pos += 1;
507                    Ok(ReceivedOption::new(parsed_code, &Vec::new()))
508                }
509                parsed_len => {
510                    let data = buf.read_vec(*pos + 2, parsed_len as usize)?;
511                    *pos += data.len() as u32 + 2;
512                    Ok(ReceivedOption::new(parsed_code, &data))
513                }
514            },
515        }
516    }
517
518    /// Reads and caches a DHCP option from the packet.
519    ///
520    /// # Parameters
521    ///
522    /// - `code` is an option code to be returned.
523    ///
524    /// # Result
525    ///
526    /// It returns unparsed DHCP option or `None` if such an option
527    /// does not exist in the packet.
528    pub fn option(&mut self, code: u8) -> Result<Option<ReceivedOption>, BufferError> {
529        // We will collect the parsed options while we search for the
530        // requested option in this vector. These options will be cached
531        // so we don't have to parse them again in case the caller asks
532        // for one of them.
533        let mut new_options = Vec::<ReceivedOption>::new();
534        // Try to find the requested option among already parsed options.
535        let result: Option<ReceivedOption> = match self.state.options.entry(code) {
536            // Cached option found. Return it.
537            Entry::Occupied(occupied_entry) => Some(occupied_entry.into_mut().clone()),
538            // Option hasn't been found yet. Let's try to find it.
539            Entry::Vacant(vacant_entry) => loop {
540                // Read next option from the packet.
541                let result = Self::try_read_option(
542                    self.state.bootp.buffer(),
543                    &mut self.state.options_cursor,
544                );
545                match result {
546                    Ok(option) => {
547                        if code == option.code {
548                            // The requested option found. Cache it and return.
549                            break Some(vacant_entry.insert(option).clone());
550                        } else {
551                            // This is not the requested option. Let's cache it
552                            // and continue searching.
553                            new_options.push(option);
554                        }
555                    }
556                    Err(err) => match err {
557                        // This error occurs when we have already gone through the
558                        // entire packet. It means that the option does not exist.
559                        // Let's return None.
560                        BufferError::ReadOutOfBounds {
561                            read_position: _,
562                            read_length: _,
563                            buffer_length: _,
564                        } => break None,
565                        // Other errors should be propagated to the caller.
566                        err => return Err(err),
567                    },
568                }
569            },
570        };
571        // If we fine some options while looking for the requested
572        // option they should be moved to the actual cache.
573        new_options.into_iter().for_each(|o| {
574            self.state.options.insert(o.code, o);
575        });
576        Ok(result)
577    }

579    /// Reads a DHCP option from the packet and transforms the returned
580    /// [`BufferError`] to [`OptionParseError`].
581    ///
582    /// It is called internally by the function parsing specific options.
583    /// These functions return the [`OptionParseError`] on error.
584    ///
585    /// # Parameters
586    ///
587    /// - `code` is an option code to be returned.
588    ///
589    /// # Result

591    /// Returns unparsed DHCP option or `None` if such an option
592    /// does not exist in the packet.
593    ///
594    /// # Errors
595    ///
596    /// Returns [`OptionParseError::BufferRead`] if there was an error
597    /// reading the option data from the packet.
598    fn option_or_parse_error(
599        &mut self,
600        code: u8,
601    ) -> Result<Option<ReceivedOption>, OptionParseError> {
602        self.option(code)
603            .map_err(|err| OptionParseError::BufferRead {
604                details: err.to_string(),

608    /// Returns parsed option 53 (DHCP Message Type).
609    ///
610    /// # Errors
611    ///
612    /// It returns [`OptionParseError::Truncated`] if the option is shorter
613    /// than one byte.
614    pub fn option_53_message_type(
615        &mut self,
616    ) -> Result<Option<OptionMessageType>, OptionParseError> {
617        match self.option_or_parse_error(OPTION_CODE_DHCP_MESSAGE_TYPE)? {
618            Some(o) => match o.data.len().cmp(&1) {
619                Ordering::Less => {
620                    // Option length should be 1. Empty option is truncated
621                    // and it carries no message type.
622                    Err(OptionParseError::Truncated {
623                        option_code: OPTION_CODE_DHCP_MESSAGE_TYPE,
624                        data_length: o.data.len(),
625                    })
626                }
627                _ => Ok(Some(OptionMessageType::new(MessageType::from(o.data[0])))),
628            },
629            _ => Ok(None),
630        }
631    }

653    /// Returns parsed option 61 (Client Identifier).
654    ///
655    /// # Errors
656    ///
657    /// It returns [`OptionParseError::Truncated`] if the option is shorter
658    /// than two bytes.
659    pub fn option_61_client_identifier(
660        &mut self,
661    ) -> Result<Option<OptionClientIdentifier>, OptionParseError> {
662        match self.option_or_parse_error(OPTION_CODE_CLIENT_IDENTIFIER)? {
663            Some(o) => match o.data.len().cmp(&2) {
664                Ordering::Less => {
665                    // Option length should be 2, including client identifier
666                    // type and the actual identifier.
667                    Err(OptionParseError::Truncated {
668                        option_code: OPTION_CODE_CLIENT_IDENTIFIER,
669                        data_length: o.data.len(),
670                    })
671                }
672                _ => Ok(Some(OptionClientIdentifier::new(
673                    HType::from(o.data[0]),
