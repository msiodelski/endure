24/// Represents errors returned by the functions storing received packets
25/// in the DHCPv4 transaction cache.
26#[derive(Debug, Error, PartialEq)]
27pub enum DHCPv4TransactionCacheError {
28    /// An error returned upon an attempt to read from the packet.
29    #[error("error parsing the packet: {details:?}")]
30    BufferRead {
31        /// Error details.
32        details: String,
33    },
34
35    /// An error returned upon an attempt to lock the packet for reading
36    /// or writing.
37    #[error("error locking the packet while updating transaction cache")]
38    PacketLock,
39
40    /// An error returned upon an attempt to insert a packet into transaction,
41    /// when the packet of this type already exists.
42    #[error("packet {message_type:?} already cached in the transaction")]
43    PacketExist {
44        /// DHCPv4 message type to which the error pertains.
45        message_type: v4::MessageType,
46    },
47
48    /// An error returned upon an attempt to insert a packet into transaction,
49    /// when the packet lacks a DHCPv4 message type.
50    #[error("packet lacks message type option")]
51    NoMessageType,
52
53    /// An error returned when trying to save a packet in the transaction when
54    /// the packet carries unsupported DHCPv4 message type.
55    #[error("message type {message_type_code:?} cannot be cached in a transaction")]
56    UnsupportedMessageType {
57        /// Message type code.
58        message_type_code: u8,
59    },
60}

62/// Indicates the type of the transaction.
63#[derive(Debug, PartialEq)]
64pub enum DHCPv4TransactionKind {
65    /// A DHCPDISCOVER/DHCPOFFER exchange.
66    ///
67    /// A boolean parameter indicates if the DHCPOFFER has been received.
68    Discovery(bool),
69    /// A 4-way exchange.
70    ///
71    /// This kind is set when a DHCPREQUEST was sent following the DHCPDISCOVER,
72    /// but neither DHCPACK or DHCPNAK.
73    FourWayExchange,
74    /// A 4-way exchange in which the server responded with DHCPACK.
75    SuccessfulFourWayExchange,
76    /// A 4-way exchange in which the server responded with DHCPNAK.
77    FailedFourWayExchange,
78    /// A renewal when client sends DHCPREQUEST without earlier sending DHCPDISCOVER.
79    Renewal,
80    /// A renewal when the server responded with DHCPACK.
81    SuccessfulRenewal,
82    /// A renewal when the server responded with DHCPNAK.
83    FailedRenewal,
84    /// A DHCPINFORM exchange case when the server has not yet responded.
85    InfRequest,
86    /// A DHCPINFORM/DHCPACK exchange.
87    SuccessfulInfRequest,
88    /// A DHCPRELEASE message sent by the client.
89    Release,
90    /// An undetermined exchange kind.
91    Undetermined,
92}

94/// A DHCPv4 transaction maintained by the [`crate::analyzer::Analyzer`].
95///
96/// It holds DHCPv4 packets grouped into matching exchanges.
97/// The packets are typically be matched by `xid`, MAC address,
98/// client identifier etc. Grouping packets allows for generating
99/// metrics such as packet processing time (the time to generate
100/// a response), number of unanswered client requests and many
101/// other.
102///
103/// In the case of the 4-way exchange, all four messages should
104/// be recorded in the [`DHCPv4Transaction`] because there is the
105/// expectation that the same `xid` is used in all messages.
106/// In case of a lease renew or information request only two
107/// messages are typically matched and set in the transaction
108/// (e.g., `DHCPREQUEST` and `DHCPACK`).
109#[derive(Debug, Clone)]
110pub struct DHCPv4Transaction {
111    /// A timestamp when the transaction instance was created.
112    pub created_at: Instant,
113    /// A DHCPDISCOVER message sent by the client.
114    pub discover: Option<TimeWrapper<SharedPartiallyParsedPacket>>,
115    /// A DHCPOFFER message sent by the server.
116    pub offer: Option<TimeWrapper<SharedPartiallyParsedPacket>>,
117    /// A DHCPREQUEST message sent by the client.
118    pub request: Option<TimeWrapper<SharedPartiallyParsedPacket>>,
119    /// A DHCPACK message sent by the server.
120    pub ack: Option<TimeWrapper<SharedPartiallyParsedPacket>>,
121    /// A DHCPNAK message sent by the server.
122    pub nak: Option<TimeWrapper<SharedPartiallyParsedPacket>>,
123    /// A DHCPINFORM message sent by the client.
124    pub inform: Option<TimeWrapper<SharedPartiallyParsedPacket>>,
125    /// A DHCPRELEASE message sent by the client.
126    pub release: Option<TimeWrapper<SharedPartiallyParsedPacket>>,
127}

150    /// Instantiates the [`DHCPv4Transaction`] and inserts first packet.
151    ///
152    /// # Parameters
153    ///
154    /// - `wrapped_packet` a received DHCP packet with a timestamp to be inserted
155    ///   into the transaction.
156    ///
157    /// # Result
158    ///
159    /// It returns the same errors as [`DHCPv4Transaction::insert`].
160    pub fn from_wrapped_packet(
161        wrapped_packet: TimeWrapper<SharedPartiallyParsedPacket>,
162    ) -> Result<Self, DHCPv4TransactionCacheError> {
163        let mut transaction = Self::new();
164        transaction.insert(wrapped_packet)?;
165        Ok(transaction)
166    }

168    /// Returns the transaction kind based on the received packets.
169    ///
170    /// The transaction that includes no packets has [`DHCPv4TransactionKind::Undetermined`]
171    /// kind. The kind changes depending on the set of packets stored
172    /// in the transaction. If the transaction includes a `DHCPDISCOVER`
173    /// packet but no `DHCPREQUEST`, it has a [`DHCPv4TransactionKind::Discovery`]
174    /// kind. It may change to [`DHCPv4TransactionKind::FourWayExchange`] when
175    /// a `DHCPREQUEST` packet is inserted. It remains [`DHCPv4TransactionKind::FourWayExchange`]
176    /// if a server response is successful (i.e., the `DHCPACK`) is received.
177    /// However, if the `DHCPNAK` is returned instead, the kind changes
178    /// to [`DHCPv4TransactionKind::FailedFourWayExchange`]. If the `DHCPINFORM`
179    /// message is recorded, and no messages normally belonging to a 4-way
180    /// exchange or a renewal, the transaction kind is [`DHCPv4TransactionKind::InfRequest`].
181    ///
182    /// The transactional auditors decide whether they should take action or
183    /// return early based on the transaction kind returned by this function.
184    pub fn kind(&self) -> DHCPv4TransactionKind {
185        match (
186            &self.discover,
187            &self.offer,
188            &self.request,
189            &self.ack,
190            &self.nak,
191            &self.inform,
192            &self.release,
193        ) {
194            (Some(_), _, None, None, None, None, None) => {
195                DHCPv4TransactionKind::Discovery(self.offer.is_some())
196            }
197            (Some(_), _, Some(_), None, None, None, None) => DHCPv4TransactionKind::FourWayExchange,
198            (Some(_), _, Some(_), Some(_), None, None, None) => {
199                DHCPv4TransactionKind::SuccessfulFourWayExchange
200            }
201            (Some(_), _, Some(_), None, Some(_), None, None) => {
202                DHCPv4TransactionKind::FailedFourWayExchange
203            }
204            (None, None, Some(_), None, None, None, None) => DHCPv4TransactionKind::Renewal,
205            (None, None, Some(_), Some(_), None, None, None) => {
206                DHCPv4TransactionKind::SuccessfulRenewal
207            }
208            (None, None, Some(_), None, Some(_), None, None) => {
209                DHCPv4TransactionKind::FailedRenewal
210            }
211            (None, None, None, None, None, Some(_), None) => DHCPv4TransactionKind::InfRequest,
212            (None, None, None, Some(_), None, Some(_), None) => {
213                DHCPv4TransactionKind::SuccessfulInfRequest
214            }
215            (_, _, _, _, _, _, Some(_)) => DHCPv4TransactionKind::Release,
216            _ => DHCPv4TransactionKind::Undetermined,
217        }
218    }
219
220    /// Inserts a packet into the transaction.
221    ///
222    /// # Parameters
223    ///
224    /// - `packet` a received DHCP packet to be inserted into the transaction.
225    ///
226    /// # Result
227    ///
228    /// If the packet with the same message type is already recorded in
229    /// the transaction it returns the [`DHCPv4TransactionCacheError::PacketExist`]
230    /// error. If the packet lacks the Message Type option, it returns the
231    /// [`DHCPv4TransactionCacheError::NoMessageType`] error. When the specified
232    /// packet is not transactional (i.e., the receiving party does not respond to
233    /// it per the protocol spec) the [`DHCPv4TransactionCacheError::UnsupportedMessageType`]
234    /// error is returned.
235    pub fn insert(
236        &mut self,
237        packet: TimeWrapper<SharedPartiallyParsedPacket>,
238    ) -> Result<(), DHCPv4TransactionCacheError> {
239        let mut locked_packet = packet
240            .get()
241            .write()
242            .map_err(|_| DHCPv4TransactionCacheError::PacketLock {})?;
243        let option_msg_type = locked_packet.option_53_message_type().map_err(|err| {
244            DHCPv4TransactionCacheError::BufferRead {
245                details: err.to_string(),
246            }
247        })?;
248        if option_msg_type.is_none() {
249            return Err(DHCPv4TransactionCacheError::NoMessageType {});
250        }
251        let msg_type = option_msg_type.unwrap().msg_type;
252        match msg_type {
253            v4::MessageType::Discover => {
254                Self::insert_or_err(&mut self.discover, msg_type, packet.clone())?
255            }
256            v4::MessageType::Offer => {
257                Self::insert_or_err(&mut self.offer, msg_type, packet.clone())?
258            }
259            v4::MessageType::Request => {
260                Self::insert_or_err(&mut self.request, msg_type, packet.clone())?
261            }
262            v4::MessageType::Nak => Self::insert_or_err(&mut self.nak, msg_type, packet.clone())?,
263            v4::MessageType::Ack => Self::insert_or_err(&mut self.ack, msg_type, packet.clone())?,
264            v4::MessageType::Inform => {
265                Self::insert_or_err(&mut self.inform, msg_type, packet.clone())?
266            }
267            v4::MessageType::Release => {
268                Self::insert_or_err(&mut self.release, msg_type, packet.clone())?
269            }
270            _ => {
271                return Err(DHCPv4TransactionCacheError::UnsupportedMessageType {
272                    message_type_code: msg_type.into(),
273                })
274            }
275        }
276        Ok(())
277    }

279    /// Conditionally sets a DHCPv4 packet instance of a given type.
280    ///
281    /// It is used internally by the [`DHCPv4Transaction::insert`] function.
282    ///
283    /// # Parameters
284    ///
285    /// - `existing_packet` is a reference to a set or unset packet instance.
286    /// - `message_type` is a DHCPv4 message type.
287    /// - `packet` is a reference to the packet to be set.
288    ///
289    /// # Results
290    ///
291    /// It returns [`DHCPv4TransactionCacheError::PacketExist`] error if the
292    /// packet has been already set.
293    fn insert_or_err(
294        existing_packet: &mut Option<TimeWrapper<SharedPartiallyParsedPacket>>,
295        message_type: MessageType,
296        new_packet: TimeWrapper<SharedPartiallyParsedPacket>,
297    ) -> Result<(), DHCPv4TransactionCacheError> {
298        if existing_packet.is_some() {
299            return Err(DHCPv4TransactionCacheError::PacketExist { message_type });
300        };
301        *existing_packet = Some(new_packet);
302        Ok(())
303    }
304}

309/// Container exposing an index for searching stored DHCPv4 transactions.

326    /// Finds an existing transaction and inserts a packet into it, or creates
327    /// a new transaction.
328    ///
329    /// # Parameters
330    ///
331    /// - `packet` is a received packet instance to be inserted into the transaction.
332    ///
333    /// # Result
334    ///
335    /// On success, it returns the transaction instance where the specified packet
336    /// belongs. It may be a new transaction or an existing transaction with other
337    /// packets in it. If the found transaction already contains a packet of
338    /// the inserted packet's type, the [`DHCPv4TransactionCacheError::PacketExist`]
339    /// error is returned. The [`DHCPv4TransactionCacheError::BufferRead`] error is
340    /// returned when parsing the packet fails. If the packet lacks the Message Type
341    /// option, the [`DHCPv4TransactionCacheError::NoMessageType`] is returned.
342    pub fn insert(
343        &mut self,
344        packet: TimeWrapper<SharedPartiallyParsedPacket>,
345    ) -> Result<DHCPv4Transaction, DHCPv4TransactionCacheError> {
346        // Get the chaddr and xid from the packet. Lock the packet only once
347        // and get both values in the and_then() function.
348        let (chaddr, xid) = packet
349            .get()
350            .clone()
351            .write()
352            .map_err(|_| DHCPv4TransactionCacheError::PacketLock {})
353            .and_then(|mut locked_packet| {
354                let chaddr = locked_packet
355                    .chaddr()
356                    .map_err(|err| DHCPv4TransactionCacheError::BufferRead {
357                        details: err.to_string(),
358                    })?
359                    .clone();
360                let xid =
361                    locked_packet
362                        .xid()
363                        .map_err(|err| DHCPv4TransactionCacheError::BufferRead {
364                            details: err.to_string(),
365                        })?;
366                Ok((chaddr, xid))
367            })?;
368
369        match self.chaddr_xid_index.entry((chaddr, xid)) {
370            Occupied(occupied_entry) => {
371                let transaction = occupied_entry.into_mut();
372                transaction.insert(packet)?;
373                Ok(transaction.clone())
374            }
375            Vacant(vacant_entry) => {
376                let transaction = DHCPv4Transaction::from_wrapped_packet(packet)?;
377                vacant_entry.insert(transaction.clone());
378                Ok(transaction)
379            }
380        }
381    }

383    /// Removes all transactions older than the specified number of seconds.
384    ///
385    /// # Parameters
386    ///
387    /// - `secs` specifies the maximum number of seconds for the transactions
388    ///   to be retained in the cache; older transactions are removed.
389    pub async fn garbage_collect_expired(&mut self, secs: i64) {
390        self.chaddr_xid_index
391            .retain(|_, transaction| transaction.created_at.elapsed() < Duration::seconds(secs));
392    }

395/// Capture and analysis profiles.
396///
397/// A profile defines a collection of auditors used for the analysis. Some
398/// of the auditors are not suitable for analyzing the capture files, others
399/// are not suitable for analyzing the live packet streams. Predefined profiles
400/// differentiate between these cases. The profiles can also select specific
401/// auditors aimed at diagnosing a certain set of issues.
402#[derive(Debug, PartialEq)]
403pub enum AuditProfile {
404    /// Enable all auditors, capture traffic from interface and analyze
405    /// using moving average window.
406    LiveStreamFull,
407    /// Enable all auditors, analyze a capture file with moving average window.
408    PcapStreamFull,
409    /// Enable all auditors, analyze a capture file and compute the metrics from
410    /// the entire capture.
411    PcapFinalFull,
412}

414/// A trait that must be implemented by auditors running checks on unparsed
415/// packets.
416///
417/// An example auditor implementing this trait is the one that gathers
418/// timestamps of the received packets (i.e., [`super::packet_time::PacketTimeAuditor`]).
419pub trait GenericPacketAuditor: Debug + Send + Sync {
420    /// Runs an audit on the received packet.
421    ///
422    /// The audit is specific to the given auditor implementing this

425    /// discard some of the packets that don't meet the audit criteria.
426    ///
427    /// # Parameters
428    ///
429    /// - `packet` - a packet wrapper holding packet metadata.
430    fn audit(&mut self, packet: &PacketWrapper);
431}

438/// A trait that must be implemented by each DHCPv4 auditor.
439///
440/// The [`crate::analyzer::Analyzer`] calls the [`DHCPv4PacketAuditor::audit`]
441/// function for each received BOOTP packet. The auditor runs specialized

448    /// trait. The auditor maintains some specific metrics gathered
449    /// from the constant analysis of the received packets. It may

461        &mut self,
462        source_ip_address: &Ipv4Addr,
463        dest_ip_address: &Ipv4Addr,

761    #[rstest]
762    #[tokio::test]
763    async fn dhcpv4_transaction_cache_garbage_collect(mut cache: CacheFixture) {
764        // Insert several DHCPDISCOVER packets into the cache with different xid values,
765        // so they are treated as different transactions.
766        for i in 0..10 {
767            _ = cache
768                .insert_with_xid(MessageType::Discover, &[i, i, i, i])
769                .expect("Failed to insert DHCPDISCOVER into cache");
770        }
771        // Initially, there are no expired transactions, so they should all
772        // remain the cache.
773        cache.inner_cache.garbage_collect_expired(100).await;
774        assert_eq!(10, cache.inner_cache.chaddr_xid_index.len());
775
776        // Expire half of the transactions.
777        cache
778            .inner_cache
779            .chaddr_xid_index
780            .iter_mut()
781            .filter(|tuple| tuple.0 .1 % 2 != 0)
782            .for_each(|tuple| {
783                tuple.1.created_at = tuple.1.created_at.sub(Duration::from_secs(100));
784            });
785        cache.inner_cache.garbage_collect_expired(80).await;
786
787        // After garbage collection, only the expired transactions should be removed.
788        assert_eq!(5, cache.inner_cache.chaddr_xid_index.len());
789    }
